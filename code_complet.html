#import library 
import pandas as pd

#read csv file
df = pd.read_csv(
"C:/Users/Anaëlle/Desktop/Stage CHU/AKAKPOVI.ANAELLE/Soutenance_Python/data/REFLEXION.csv",
sep=","
)

#print results
print(df.head())
print(df.shape)
print(df.columns[:10])
print(df["sid"])

#remove index line 3 and reset/reinitialize index
df = df.drop(df.index[3]).reset_index(drop=True)

# select columns with "css" 
df_css = df.filter(regex='css', axis=1)

# show first lines
print(df_css.head())
# Remove columns with "rea" and columns finish by "css1"
df_css_clean = df_css.filter(regex='^(?!.*rea)(?!.*css1$)', axis=1)


# Keep columns that start with "v"              
df_v = df_css_clean.filter(regex='^v', axis=1)

# Remove columns "v" with cssforce1, cssforce2, cssforce3 or cssforcescref
df_v = df_v.filter(regex='^(?!.*cssforce([123]|scref)).*', axis=1)

# Reset index
df_v = df_v.reset_index(drop=True)

# Keep 3 columns "icsstemp_s", "icsssud_s", "icssforcemoy_s"
df_exceptions = df_css_clean.filter(items=["icsstemp_s", "icsssud_s", "icssforcemoy_s"])
df_exceptions = df_exceptions.reset_index(drop=True)

# Combine
df_css_clean = pd.concat([df_v, df_exceptions], axis=1)

#see results
print(df_css_clean.shape)
print(df_css_clean.columns)
print(df_css_clean.head())


#see clean table
df_css_clean
# New DataFrame for binarisation
binarisation = df_css_clean.copy()

# Select columns with "cssdoul"
colonnes_cssdoul = binarisation.filter(regex='cssdoul', axis=1).columns

# Apply binairy transformation : >=8 -> 1, <8 -> 0
binarisation[colonnes_cssdoul] = binarisation[colonnes_cssdoul].applymap(
    lambda x: 1 if x >= 8 else (0 if pd.notna(x) else pd.NA)
)
# Preview
print("Colonnes transformées en binaire :", list(colonnes_cssdoul))
print(binarisation[colonnes_cssdoul].head())
#csstemp already binarize
colonnes_csstemp = df_css_clean.filter(regex='csstemp', axis=1).columns
print(list(colonnes_csstemp))
# View five first lines csstemp
print(df_css_clean[colonnes_csstemp].head())
# Select columns with "csssud" to check values
colonnes_csssud = df_css_clean.filter(regex='csssud', axis=1).columns
print("Colonnes csssud :", list(colonnes_csssud))
# View five first lines
print(df_css_clean[colonnes_csssud].head())
#check data for "cssoed" already binarized

#Columns with cssoed

#check data for "forcemoy" 

# Columns "cssforcemoy"

print("Colonnes cssforcemoy :", list(colonnes_cssforcemoy))

#view results
print("
Aperçu des colonnes cssforcemoy :")
print(df_css_clean[colonnes_cssforcemoy].head())


#binarisation "sudation" and "force moyenne" 
# pathologic columns "csssud" 30% difference
colonnes_csssud = df_css_clean.filter(regex='^v.*_csssud$', axis=1).columns

# Binarisation 
binarisation_sud = ((df_css_clean[colonnes_csssud].sub(df_css_clean['icsssud_s'], axis=0)
                     .abs()
                     .div(df_css_clean['icsssud_s'], axis=0)) >= 0.3)

binarisation_sud = binarisation_sud.where(df_css_clean[colonnes_csssud].notna()).astype("Int64")

# view
print(binarisation_sud.head())

# pathologic columns "forcemoy"
colonnes_cssforcemoy = df_css_clean.filter(regex='^v.*_cssforcemoy$', axis=1).columns

# Binarisation 5kg difference
binarisation_forcemoy = ((df_css_clean[colonnes_cssforcemoy]
                          .sub(df_css_clean['icssforcemoy_s'], axis=0)
                          .abs()) >= 5)

binarisation_forcemoy = binarisation_forcemoy.where(df_css_clean[colonnes_cssforcemoy].notna()).astype("Int64")
# View
print(binarisation_forcemoy.head()) 
# Final table based on df_css_clean
df_binarise_final = df_css_clean.copy()

#replace pain columns by binary version 
df_binarise_final[colonnes_cssdoul] = binarisation[colonnes_cssdoul]

#replace "sudation" column by binary version
df_binarise_final[colonnes_csssud] = binarisation_sud

#replace "forcemoy" columns
df_binarise_final[colonnes_cssforcemoy] = binarisation_forcemoy

#delete healthy columns
df_binarise_final = df_binarise_final.drop(
    columns=["icsssud_s", "icssforcemoy_s", "icsstemp_s"],
    errors="ignore"
)
